<script>
    // Interactive checkbox functionality
    const checkboxes = document.querySelectorAll('.checkbox');
    const progressBar = document.getElementById('progressBar');
    const checkedCount = document.getElementById('checkedCount');
    const totalCount = document.getElementById('totalCount');

    // Set total count
    totalCount.textContent = checkboxes.length;

    // Update progress
    function updateProgress() {
        const checked = document.querySelectorAll('.checkbox:checked').length;
        const percentage = (checked / checkboxes.length) * 100;
        
        progressBar.style.width = percentage + '%';
        checkedCount.textContent = checked;

        // Update parent item styling
        checkboxes.forEach(checkbox => {
            const item = checkbox.closest('.checklist-item');
            if (checkbox.checked) {
                item.classList.add('checked');
            } else {
                item.classList.remove('checked');
            }
        });

        // Save to localStorage
        const checkedItems = Array.from(checkboxes).map(cb => cb.checked);
        localStorage.setItem('cryptoTaxChecklist', JSON.stringify(checkedItems));
    }

    // Load saved state
    const savedState = localStorage.getItem('cryptoTaxChecklist');
    if (savedState) {
        const checkedItems = JSON.parse(savedState);
        checkboxes.forEach((checkbox, index) => {
            if (checkedItems[index]) {
                checkbox.checked = true;
            }
        });
    }

    // Add event listeners
    checkboxes.forEach(checkbox => {
        checkbox.addEventListener('change', updateProgress);
    });

    // Initial progress update
    updateProgress();

    // Add click handler to entire item
    document.querySelectorAll('.checklist-item').forEach(item => {
        item.addEventListener('click', (e) => {
            if (e.target.type !== 'checkbox') {
                const checkbox = item.querySelector('.checkbox');
                checkbox.checked = !checkbox.checked;
                updateProgress();
            }
        });
    });
</script>
