                const tickerContainer = document.getElementById('stock-ticker-content');
                const sp500Container = document.getElementById('sp500-data');
                const nasdaqContainer = document.getElementById('nasdaq-data');
                const btcContainer = document.getElementById('btc-data');
                const ethContainer = document.getElementById('eth-data');

                if (FMP_API_KEY === 'YOUR_FINANCIAL_MODELING_PREP_API_KEY') {
                    console.warn("FMP API Key is a placeholder. Market data will not be fetched.");
                    return;
                }

                try {
                    const response = await fetch(`https://financialmodelingprep.com/api/v3/quote/%5EGSPC,%5EIXIC,BTCUSD,ETHUSD?apikey=${FMP_API_KEY}`);
                    const data = await response.json();
                    
                    if (Array.isArray(data)) {
                        data.forEach(item => {
                            const changeClass = item.changesPercentage >= 0 ? 'text-green-400' : 'text-red-400';
                            const html = `<div><h3 class="text-sm uppercase text-gray-400">${item.name}</h3><p class="text-2xl font-bold">${item.price.toFixed(2)}</p><p class="${changeClass}">${item.change.toFixed(2)} (${item.changesPercentage.toFixed(2)}%)</p></div>`;
                            if (item.symbol === '^GSPC') sp500Container.innerHTML = html;
                            if (item.symbol === '^IXIC') nasdaqContainer.innerHTML = html;
                            if (item.symbol === 'BTCUSD') btcContainer.innerHTML = html;
                            if (item.symbol === 'ETHUSD') ethContainer.innerHTML = html;
                        });
                    }

                    const tickerResponse = await fetch(`https://financialmodelingprep.com/api/v3/quote/AAPL,MSFT,GOOGL,AMZN,TSLA,NVDA?apikey=${FMP_API_KEY}`);
                    const tickerData = await tickerResponse.json();
                    
                    if (Array.isArray(tickerData)) {
                        let tickerHtml = '';
                        tickerData.forEach(stock => {
                            const changeClass = stock.changesPercentage >= 0 ? 'text-green-400' : 'text-red-400';
                            tickerHtml += `<div class="ticker-item"><span class="font-bold">${stock.symbol}</span><span class="ml-2">${stock.price.toFixed(2)}</span><span class="ml-2 ${changeClass}">${stock.changesPercentage.toFixed(2)}%</span></div>`;
                        });
                        tickerContainer.innerHTML = tickerHtml + tickerHtml;
                    }

                } catch (error) {
                    console.error('Error fetching market data:', error);
                    if(tickerContainer) tickerContainer.innerHTML = '<div class="ticker-item">Market data currently unavailable.</div>';
                }
            }

            async function fetchFinancialNews() {
                const newsContainer = document.getElementById('news-container');
                if (MARKETAUX_API_KEY === 'YOUR_MARKETAUX_API_KEY') {
                    console.warn("MarketAux API Key is a placeholder. News will not be fetched.");
                    if(newsContainer) newsContainer.innerHTML = '<p>News feed is not configured. Please add an API key.</p>';
                    return;
                }
                try {
                    const response = await fetch(`https://api.marketaux.com/v1/news/all?symbols=TSLA,AMZN,MSFT&filter_entities=true&language=en&api_token=${MARKETAUX_API_KEY}`);
                    const newsData = await response.json();
                    
                    if (newsData && Array.isArray(newsData.data)) {
                        let newsHtml = '';
                        newsData.data.slice(0, 5).forEach(article => {
                            newsHtml += `<div class="p-4 border rounded-lg hover:shadow-md transition-shadow"><a href="${article.url}" target="_blank" rel="noopener noreferrer"><h3 class="text-xl font-bold mb-2">${article.title}</h3><p class="text-gray-600 text-sm mb-2">${article.snippet}</p><p class="text-xs text-gray-400">Source: ${article.source}</p></a></div>`;
                        });
                        newsContainer.innerHTML = newsHtml;
                    } else {
                        console.error('News data response is not in the expected format:', newsData);
                        newsContainer.innerHTML = '<p>Could not load financial news. Check API key or response format.</p>';
                    }
                } catch (error) {
                    console.error('Error fetching financial news:', error);
                    if(newsContainer) newsContainer.innerHTML = '<p>Could not load financial news at this time.</p>';
                }
            }
