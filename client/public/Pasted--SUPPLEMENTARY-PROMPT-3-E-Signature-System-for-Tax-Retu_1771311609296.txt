# SUPPLEMENTARY PROMPT #3: E-Signature System for Tax Returns

## Overview

Build a legally compliant electronic signature system that allows clients to review and sign their tax returns digitally, with full audit trails and compliance with IRS e-signature requirements and ESIGN Act.

-----

## Legal Requirements & Compliance

### IRS E-Signature Requirements

- Must capture client’s intent to sign
- Must authenticate signer identity
- Must create tamper-evident audit trail
- Must preserve signature for IRS retention period (3+ years)
- Must include timestamp and IP address
- Must obtain explicit consent to sign electronically

### Canvas-Based Signature Pad

```jsx
import SignatureCanvas from 'react-signature-canvas'

function SignaturePad() {
  const sigCanvas = useRef(null)
  const [signatureData, setSignatureData] = useState(null)
  const [isSigned, setIsSigned] = useState(false)
  
  const clearSignature = () => {
    sigCanvas.current.clear()
    setSignatureData(null)
    setIsSigned(false)
  }
  
  const saveSignature = () => {
    if (sigCanvas.current.isEmpty()) {
      alert('Please provide a signature first')
      return
    }
    
    const dataURL = sigCanvas.current.toDataURL('image/png')
    setSignatureData(dataURL)
    setIsSigned(true)
  }
  
  return (
    <SignatureContainer>
      <SignatureInstructions>
        <h3>Sign Below</h3>
        <p>Use your finger or mouse to sign in the box below</p>
      </SignatureInstructions>
      
      <SignatureCanvasWrapper>
        <SignatureCanvas
          ref={sigCanvas}
          penColor="black"
          canvasProps={{
            width: 500,
            height: 200,
            className: 'signature-canvas'
          }}
        />
      </SignatureCanvasWrapper>
      
      <ButtonGroup>
        <Button onClick={clearSignature} variant="secondary">
          Clear
        </Button>
        <Button onClick={saveSignature} variant="primary">
          Save Signature
        </Button>
      </ButtonGroup>
      
      {isSigned && (
        <SignaturePreview>
          <h4>Your Signature:</h4>
          <img src={signatureData} alt="Signature preview" />
        </SignaturePreview>
      )}
    </SignatureContainer>
  )
}
```

```jsx
function SignatureCollectionPage({ returnId }) {
  const [signature, setSignature] = useState(null)
  const [taxpayerName, setTaxpayerName] = useState('')
  const [spouseSignature, setSpouseSignature] = useState(null)
  const [spouseName, setSpouseName] = useState('')
  const [isJointReturn, setIsJointReturn] = useState(false)
  const [isSubmitting, setIsSubmitting] = useState(false)
  
  const handleSubmitSignature = async () => {
    if (!signature) {
      alert('Please provide your signature')
      return
    }
    
    if (!taxpayerName.trim()) {
      alert('Please type your full name')
      return
    }
    
    if (isJointReturn && (!spouseSignature || !spouseName.trim())) {
      alert('Spouse signature required for joint return')
      return
    }
    
    setIsSubmitting(true)
    
    try {
      const signatureData = {
        returnId,
        taxpayerSignature: signature,
        taxpayerName: taxpayerName.trim(),
        spouseSignature: isJointReturn ? spouseSignature : null,
        spouseName: isJointReturn ? spouseName.trim() : null,
        signedAt: new Date().toISOString(),
        ipAddress: await getClientIP(),
        userAgent: navigator.userAgent,
        deviceType: getMobileOperatingSystem()
      }
      
      const response = await fetch('/api/signatures/submit', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${getToken()}`
        },
        body: JSON.stringify(signatureData)
      })
      
      if (response.ok) {
        // Show success message
        showSuccessMessage()
        // Redirect to confirmation page
        router.push(`/signature-complete/${returnId}`)
      } else {
        throw new Error('Signature submission failed')
      }
    } catch (error) {
      alert('Error submitting signature. Please try again.')
      console.error(error)
    } finally {
      setIsSubmitting(false)
    }
  }
  
  return (
    <SignaturePageContainer>
      <SignatureHeader>
        <h1>Sign Your Tax Return</h1>
        <p>Your signature confirms that you have reviewed and approve your tax return</p>
      </SignatureHeader>
      
      <SignatureSection>

          Date: {new Date().toLocaleDateString()}

          <h2>Spouse Signature</h2>
          
          <FormField>
            <label>Spouse's Full Legal Name</label>

              type="text"

          </FormField>

            onSave={(sig) => setSpouseSignature(sig)}

            Date: {new Date().toLocaleDateString()}

  )
```

```javascript
router.post('/api/signatures/submit', authenticateUser, async (req, res) => {
  try {
    const {
      returnId,
      taxpayerSignature,
      taxpayerName,
      spouseSignature,
      spouseName,
      signedAt,
      ipAddress,
      userAgent,
      deviceType
    } = req.body
    
    const userId = req.user.id
    
    // Validate return belongs to user
    const taxReturn = await db.taxReturns.findFirst({
      where: { id: returnId, userId }
    })

      return res.status(404).json({ error: 'Tax return not found' })
    }
    
    // Create signature record
    const signature = await db.signatures.create({
      data: {
        returnId,
        userId,
        taxpayerSignature,
        taxpayerName,
        spouseSignature,
        spouseName,
        signedAt: new Date(signedAt),
        ipAddress,
        userAgent,
        deviceType,
        signatureHash: generateSignatureHash(taxpayerSignature)
      }
    })
    
    // Update return status
    await db.taxReturns.update({
      where: { id: returnId },
      data: { 
        status: 'signed',
        signedAt: new Date()
      }
    })
    
    // Generate signed PDF with signatures embedded
    await embedSignaturesInPDF(taxReturn.filePath, signature)
    
    // Send confirmation email
    await sendSignatureConfirmationEmail(userId, returnId)
    
    // Notify admin
    await notifyAdminOfSignedReturn(userId, returnId)
    
    res.json({ success: true, signatureId: signature.id })
  } catch (error) {
    console.error('Signature submission error:', error)
    res.status(500).json({ error: 'Failed to process signature' })
  }
})
```

### Embed Signatures in PDF

```javascript
const { PDFDocument, rgb } = require('pdf-lib')
const fs = require('fs').promises

async function embedSignaturesInPDF(pdfPath, signatureData) {
  // Load the PDF
  const existingPdfBytes = await fs.readFile(pdfPath)
  const pdfDoc = await PDFDocument.load(existingPdfBytes)
  
  // Get signature page (usually last page)
  const pages = pdfDoc.getPages()
  const signaturePage = pages[pages.length - 1]
  
  // Convert base64 signature to image
  const taxpayerSigImage = await pdfDoc.embedPng(
    Buffer.from(signatureData.taxpayerSignature.split(',')[1], 'base64')
  )
  
  // Add taxpayer signature
  signaturePage.drawImage(taxpayerSigImage, {
    x: 100,
    y: 150,
    width: 200,
    height: 50
  })
  
  // Add taxpayer name
  signaturePage.drawText(signatureData.taxpayerName, {
    x: 100,
    y: 140,
    size: 10,
    color: rgb(0, 0, 0)
  })
  
  // Add date
  signaturePage.drawText(new Date(signatureData.signedAt).toLocaleDateString(), {
    x: 310,
    y: 150,
    size: 10,
    color: rgb(0, 0, 0)
  })
  
  // If joint return, add spouse signature
  if (signatureData.spouseSignature) {
    const spouseSigImage = await pdfDoc.embedPng(
      Buffer.from(signatureData.spouseSignature.split(',')[1], 'base64')
    )
    
    signaturePage.drawImage(spouseSigImage, {
      x: 100,
      y: 80,
      width: 200,
      height: 50
    })
    
    signaturePage.drawText(signatureData.spouseName, {
      x: 100,
      y: 70,
      size: 10,
      color: rgb(0, 0, 0)
    })
    
    signaturePage.drawText(new Date(signatureData.signedAt).toLocaleDateString(), {
      x: 310,
      y: 80,
      size: 10,
      color: rgb(0, 0, 0)
    })
  }
  
  // Add metadata
  pdfDoc.setTitle(`Signed Tax Return - ${signatureData.taxpayerName}`)
  pdfDoc.setAuthor('Tax Portal')
  pdfDoc.setSubject('Electronically Signed Tax Return')
  pdfDoc.setCreationDate(new Date(signatureData.signedAt))
  
  // Save signed PDF
  const signedPdfBytes = await pdfDoc.save()
  const signedPdfPath = pdfPath.replace('.pdf', '-signed.pdf')
  await fs.writeFile(signedPdfPath, signedPdfBytes)
  
  // Update database with signed PDF path
  await db.taxReturns.update({
    where: { id: signatureData.returnId },
    data: { signedFilePath: signedPdfPath }
  })
  
  return signedPdfPath
}
```

### Tamper Detection

```javascript
const crypto = require('crypto')

function generateSignatureHash(signatureDataURL) {
  return crypto
    .createHash('sha256')
    .update(signatureDataURL)
    .digest('hex')
}

async function verifySignatureIntegrity(signatureId) {
  const signature = await db.signatures.findUnique({
    where: { id: signatureId }
  })
  
  const currentHash = generateSignatureHash(signature.taxpayerSignature)
  
  if (currentHash !== signature.signatureHash) {
    // Signature has been tampered with!
    await db.securityAlerts.create({
      data: {
        alertType: 'SIGNATURE_TAMPERING',
        signatureId,
        detectedAt: new Date(),
        severity: 'CRITICAL'
      }
    })
    
    return false
  }
  
  return true
}
```
