    <script>
        let chart;

        function formatCurrency(num) {
            return new Intl.NumberFormat('en-US', { 
                style: 'currency', 
                currency: 'USD',
                maximumFractionDigits: 0 
            }).format(num);
        }

        function calculate() {
            const currentAge = parseInt(document.getElementById('current-age').value) || 0;
            const retireAge = parseInt(document.getElementById('retire-age').value) || 0;
            const currentAssets = parseFloat(document.getElementById('current-assets').value) || 0;
            const monthlyContribution = parseFloat(document.getElementById('monthly-contribution').value) || 0;
            const returnRate = (parseFloat(document.getElementById('return-rate').value) || 0) / 100;
            const desiredIncomeToday = parseFloat(document.getElementById('desired-income').value) || 0;
            const inflationRate = (parseFloat(document.getElementById('inflation-rate').value) || 0) / 100;
            const safeWithdrawalRate = (parseFloat(document.getElementById('withdrawal-rate').value) || 0) / 100;

            const yearsToRetirement = Math.max(0, retireAge - currentAge);
            
            // 1. Calculate Target Nest Egg in Future Dollars
            // First, adjust desired income for inflation
            const futureAnnualIncome = desiredIncomeToday * Math.pow(1 + inflationRate, yearsToRetirement);
            // Required nest egg = Annual Income / SWR
            const targetNestEgg = futureAnnualIncome / safeWithdrawalRate;

            // 2. Project Future Value of Assets
            let labels = [];
            let assetData = [];
            let targetLineData = [];
            let currentBalance = currentAssets;
            const monthlyRate = returnRate / 12;

            labels.push(`Age ${currentAge}`);
            assetData.push(currentBalance);
            targetLineData.push(null); // Target only matters at the end

            for (let year = 1; year <= yearsToRetirement; year++) {
                // Compound monthly contributions
                for (let month = 1; month <= 12; month++) {
                    currentBalance = (currentBalance + monthlyContribution) * (1 + monthlyRate);
                }
                labels.push(`Age ${currentAge + year}`);
                assetData.push(currentBalance);
                targetLineData.push(null);
            }
            
            // Set the target point for the last year
            targetLineData[targetLineData.length - 1] = targetNestEgg;

            const projectedFinalSavings = currentBalance;
            const readinessPercent = Math.min(100, Math.round((projectedFinalSavings / targetNestEgg) * 100)) || 0;

            // Update UI
            document.getElementById('target-nest-egg').innerText = formatCurrency(targetNestEgg);
            document.getElementById('projected-savings').innerText = formatCurrency(projectedFinalSavings);
            document.getElementById('readiness-score').innerText = readinessPercent + '%';
            document.getElementById('readiness-bar').style.width = readinessPercent + '%';

            // Insights logic
            let insight = "";
            if (readinessPercent >= 100) {
                insight = `Congratulations! You are on track to retire at ${retireAge}. In fact, your projected savings of ${formatCurrency(projectedFinalSavings)} exceeds your target of ${formatCurrency(targetNestEgg)}. You might even consider retiring a few years earlier!`;
                document.getElementById('insight-box').className = "glass p-6 border-l-8 border-emerald-600 bg-emerald-50/50";
            } else {
                const gap = targetNestEgg - projectedFinalSavings;
                insight = `You have a gap of ${formatCurrency(gap)} to reach your retirement goal. To bridge this, you could increase your monthly contributions, aim for a higher investment return, or push your retirement age by a few years.`;
                document.getElementById('insight-box').className = "glass p-6 border-l-8 border-orange-600 bg-orange-50/50";
            }
            document.getElementById('insight-text').innerText = insight;

            updateChart(labels, assetData, targetNestEgg);
        }

        function updateChart(labels, data, target) {
            const ctx = document.getElementById('projectionChart').getContext('2d');
            
            if (chart) chart.destroy();

            chart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [
                        {
                            label: 'Projected Wealth',
                            data: data,
                            borderColor: '#3b82f6',
                            backgroundColor: 'rgba(59, 130, 246, 0.1)',
                            fill: true,
                            tension: 0.4,
                            pointRadius: 0,
                            pointHoverRadius: 6,
                            borderWidth: 4
                        },
                        {
                            label: 'Target Goal',
                            data: labels.map((_, i) => i === labels.length - 1 ? target : null),
                            borderColor: '#f59e0b',
                            backgroundColor: '#f59e0b',
                            pointStyle: 'star',
                            pointRadius: 10,
                            showLine: false
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    interaction: { intersect: false, mode: 'index' },
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    return context.dataset.label + ': ' + formatCurrency(context.parsed.y);
                                }
                            }
                        }
                    },
                    scales: {
                        y: {
                            beginAtZero: true,
                            ticks: { callback: (value) => formatCurrency(value) },
                            grid: { color: 'rgba(0,0,0,0.05)' }
                        },
                        x: {
                            grid: { display: false }
                        }
                    }
                }
            });
        }

        // Event listeners
        const allInputs = document.querySelectorAll('input');
        allInputs.forEach(input => {
            input.addEventListener('input', calculate);
        });

        window.onload = calculate;
    </script>
