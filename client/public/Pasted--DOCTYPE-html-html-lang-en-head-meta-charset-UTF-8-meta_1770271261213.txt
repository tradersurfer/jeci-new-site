    <script>
        let allocChart;

        function formatCurrency(n) {
            return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 }).format(n);
        }

        function calculate() {
            let totalLiquid = 0;
            let totalIlliquid = 0;
            let totalDebt = 0;

            // Calculate Assets
            document.querySelectorAll('.asset-input').forEach(input => {
                const val = parseFloat(input.value) || 0;
                if(input.dataset.category === 'liquid') totalLiquid += val;
                else totalIlliquid += val;
            });

            // Calculate Liabilities
            document.querySelectorAll('.debt-input').forEach(input => {
                totalDebt += parseFloat(input.value) || 0;
            });

            const totalAssets = totalLiquid + totalIlliquid;
            const netWorth = totalAssets - totalDebt;

            // Update Metrics
            document.getElementById('total-net-worth').innerText = formatCurrency(netWorth);
            
            const liqRatio = totalAssets > 0 ? (totalLiquid / totalAssets) * 100 : 0;
            document.getElementById('metric-liquidity').innerText = liqRatio.toFixed(1) + '%';
            
            const debtRatio = totalAssets > 0 ? (totalDebt / totalAssets) * 100 : 0;
            document.getElementById('metric-dta').innerText = debtRatio.toFixed(1) + '%';

            // Concentration Risk logic
            const maxAsset = Math.max(totalLiquid, totalIlliquid); // Simple grouping check
            const conPercent = totalAssets > 0 ? (maxAsset / totalAssets) * 100 : 0;
            const conMetric = document.getElementById('metric-concentration');
            
            if(conPercent > 70) {
                conMetric.innerText = "CRITICAL";
                conMetric.className = "text-2xl font-black text-red-600";
            } else if(conPercent > 50) {
                conMetric.innerText = "HIGH";
                conMetric.className = "text-2xl font-black text-orange-500";
            } else {
                conMetric.innerText = "DIVERSIFIED";
                conMetric.className = "text-2xl font-black text-emerald-600";
            }
            document.getElementById('concentration-desc').innerText = `Largest group is ${conPercent.toFixed(0)}% of assets`;

            // Update Progress Bars
            document.getElementById('liquid-total-text').innerText = formatCurrency(totalLiquid);
            document.getElementById('illiquid-total-text').innerText = formatCurrency(totalIlliquid);
            document.getElementById('liquid-bar').style.width = liqRatio + '%';
            document.getElementById('illiquid-bar').style.width = (totalAssets > 0 ? (totalIlliquid/totalAssets)*100 : 0) + '%';

            updateTable(totalLiquid, totalIlliquid, totalDebt, totalAssets, netWorth);
            updateChart(totalLiquid, totalIlliquid);
        }

        function updateTable(liq, illiq, debt, assets, nw) {
            const tbody = document.getElementById('summary-table-body');
            const data = [
                { name: 'Liquid Assets', val: liq, p: (liq/assets*100), color: 'text-blue-600', status: 'Available' },
                { name: 'Illiquid Assets', val: illiq, p: (illiq/assets*100), color: 'text-emerald-600', status: 'Tied Up' },
                { name: 'Total Liabilities', val: debt, p: null, color: 'text-red-500', status: 'Owed' }
            ];

            tbody.innerHTML = data.map(item => `
                <tr class="border-b border-slate-50">
                    <td class="py-4 font-semibold text-slate-700">${item.name}</td>
                    <td class="py-4 font-mono font-bold">${formatCurrency(item.val)}</td>
                    <td class="py-4 text-slate-400 font-medium">${item.p ? item.p.toFixed(1) + '%' : '-'}</td>
                    <td class="py-4 text-right"><span class="px-2 py-1 rounded-md text-[10px] font-bold uppercase bg-slate-100 ${item.color}">${item.status}</span></td>
                </tr>
            `).join('');
        }

        function updateChart(liq, illiq) {
            const ctx = document.getElementById('allocationChart').getContext('2d');
            if(allocChart) allocChart.destroy();

            allocChart = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Liquid Portfolio', 'Illiquid / Business'],
                    datasets: [{
                        data: [liq, illiq],
                        backgroundColor: ['#3b82f6', '#10b981'],
                        hoverOffset: 15,
                        borderWidth: 0
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'bottom', labels: { usePointStyle: true, padding: 20, font: { weight: '700' } } }
                    },
                    cutout: '70%'
                }
            });
        }

        // Listeners
        document.querySelectorAll('input').forEach(input => {
            input.addEventListener('input', calculate);
        });

        window.onload = calculate;
    </script>
