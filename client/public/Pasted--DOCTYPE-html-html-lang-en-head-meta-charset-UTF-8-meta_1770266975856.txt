            function formatCurrency(value) {
                if (isNaN(value) || !isFinite(value)) return 'N/A';
                const formatted = value.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
                return `$${formatted}`;

            function formatPercentage(value) {

                balanceSheetData.cash.push(currentAssumptions.initialCashBalance);
                balanceSheetData.debt.push(currentAssumptions.initialDebtBalance);
                balanceSheetData.equity.push(currentAssumptions.initialCashBalance - currentAssumptions.initialDebtBalance); // Simple initial equity
                balanceSheetData.ppe.push(0); // Initial PPE

                // --- Monthly Calculation Loop ---
                for (let month = 0; month < totalMonths; month++) {
                    const year = Math.floor(month / monthsPerYear);
                    const isNewYear = (month % monthsPerYear === 0 && month !== 0);

                    // Apply annual increases at the start of each new year
                    if (isNewYear) {
                        currentRentMonthly *= (1 + currentAssumptions.rentIncreaseAnnual);
                        currentAverageSalaryAnnual *= (1 + currentAssumptions.salaryIncreaseAnnual);
                        currentPricingMultiplier *= (1 + currentAssumptions.pricingIncreaseAnnual);
                        currentHeadcount *= (1 + currentAssumptions.headcountGrowthAnnual);
                    }

                    // --- Revenue Calculation ---
                    // New customers added this month
                    let newCustomersThisMonth = currentAssumptions.monthlyNewCustomers;
                    // Customers lost this month
                    let churnedCustomers = currentCustomers * currentAssumptions.churnRate;
                    // Update active customers
                    currentCustomers = currentCustomers + newCustomersThisMonth - churnedCustomers;
                    // Ensure customers don't go below 0
                    currentCustomers = Math.max(0, currentCustomers);

                    let monthlyRevenue = currentCustomers * currentAssumptions.averageOrderValue * currentPricingMultiplier;
                    monthlyData.revenue.push(monthlyRevenue);

                    // --- COGS ---
                    let unitsSold = currentCustomers; // Simplified: 1 unit per customer
                    let monthlyCOGS = unitsSold * currentAssumptions.variableCostPerUnit;
                    monthlyData.cogs.push(monthlyCOGS);

                    let grossProfit = monthlyRevenue - monthlyCOGS;
                    monthlyData.grossProfit.push(grossProfit);

                    // --- Operating Expenses ---
                    let marketingExpenses = monthlyRevenue * currentAssumptions.marketingSpendPercentRevenue;
                    monthlyData.marketingExpenses.push(marketingExpenses);

                    let salaryExpenses = (currentHeadcount * currentAverageSalaryAnnual) / monthsPerYear; // Monthly salary
                    monthlyData.salaryExpenses.push(salaryExpenses);

                    monthlyData.rentExpenses.push(currentRentMonthly);
                    monthlyData.otherOperatingExpenses.push(currentAssumptions.otherFixedOperatingCostsMonthly);

                    let totalOperatingExpenses = marketingExpenses + salaryExpenses + currentRentMonthly + currentAssumptions.otherFixedOperatingCostsMonthly;
                    monthlyData.totalOperatingExpenses.push(totalOperatingExpenses);

                    // --- Depreciation (Annual CapEx depreciated monthly) ---
                    let monthlyCapex = 0;
                    if (isNewYear) {
                        monthlyCapex = currentAssumptions.capexAnnual;
                    }
                    monthlyData.capexActual.push(monthlyCapex); // Store actual CapEx for cash flow

                    // Add new CapEx to PPE
                    currentPPE += monthlyCapex;

                    // Calculate monthly depreciation based on current PPE (simplified straight-line)
                    let monthlyDepreciation = (currentPPE * currentAssumptions.depreciationRateAnnual) / monthsPerYear;
                    monthlyData.depreciation.push(monthlyDepreciation);
                    currentAccumulatedDepreciation += monthlyDepreciation;

                    // --- P&L Calculation ---
                    let ebit = grossProfit - totalOperatingExpenses - monthlyDepreciation; // Operating Income
                    monthlyData.ebit.push(ebit);

                    // Interest Expense (on current debt balance)
                    let monthlyInterestRate = currentAssumptions.interestRateAnnual / monthsPerYear;
                    let interestExpense = currentDebtBalance * monthlyInterestRate;
                    monthlyData.interestExpense.push(interestExpense);

                    let ebt = ebit - interestExpense;
                    monthlyData.ebt.push(ebt);

                    let taxes = ebt > 0 ? ebt * currentAssumptions.taxRate : 0;
                    monthlyData.taxes.push(taxes);

                    let netProfit = ebt - taxes;
                    monthlyData.netProfit.push(netProfit);

                    // --- Cash Flow Statement Calculation ---
                    // Operating Activities
                    let netIncomeCF = netProfit;
                    cashFlowData.netIncomeCF.push(netIncomeCF);
                    cashFlowData.depreciationCF.push(monthlyDepreciation);
                    cashFlowData.changeInAR.push(0); // Simplified: No AR changes
                    cashFlowData.changeInAP.push(0); // Simplified: No AP changes
                    let netCashFromOperating = netIncomeCF + monthlyDepreciation; // Simplified: no working capital changes
                    cashFlowData.netCashFromOperating.push(netCashFromOperating);

                    // Investing Activities
                    let capexCF = -monthlyCapex; // CapEx is a cash outflow
                    let currentMonthNetCashFromInvesting = capexCF; // Local variable for clarity
                    cashFlowData.capexCF.push(capexCF);
                    cashFlowData.netCashFromInvesting.push(currentMonthNetCashFromInvesting); // Push the value to the array

                    // Financing Activities
                    let debtIssuanceRepaymentCF = 0; // Simplified: No debt changes unless specified
                    cashFlowData.debtIssuanceRepaymentCF.push(debtIssuanceRepaymentCF);
                    let ownerDrawsCF = 0; // Will be populated by profit allocation
                    cashFlowData.ownerDrawsCF.push(ownerDrawsCF); // Placeholder for now

                    let netCashFromFinancing = debtIssuanceRepaymentCF + ownerDrawsCF;
                    cashFlowData.netCashFromFinancing.push(netCashFromFinancing);

                    let netChangeInCash = netCashFromOperating + currentMonthNetCashFromInvesting + netCashFromFinancing;
                    cashFlowData.netChangeInCash.push(netChangeInCash);

                    let beginningCash = cashFlowData.endingCash[month - 1] || currentAssumptions.initialCashBalance;
                    cashFlowData.beginningCash.push(beginningCash);
                    let endingCash = beginningCash + netChangeInCash;
                    cashFlowData.endingCash.push(endingCash);

                    // --- Balance Sheet Calculation ---
                    // Assets
                    balanceSheetData.cash.push(endingCash);
                    balanceSheetData.accountsReceivable.push(0); // Simplified
                    balanceSheetData.ppe.push(currentPPE - currentAccumulatedDepreciation); // Net PPE
                    let totalAssets = endingCash + (currentPPE - currentAccumulatedDepreciation); // Simplified
                    balanceSheetData.totalAssets.push(totalAssets);

                    // Liabilities
                    balanceSheetData.accountsPayable.push(0); // Simplified
                    balanceSheetData.debt.push(currentDebtBalance);
                    let totalLiabilities = currentDebtBalance; // Simplified
                    balanceSheetData.totalLiabilities.push(totalLiabilities);

                    // Equity
                    let beginningEquity = balanceSheetData.equity[month - 1] || (currentAssumptions.initialCashBalance - currentAssumptions.initialDebtBalance);
                    let currentEquity = beginningEquity + netProfit - ownerDrawsCF; // Simplified
                    balanceSheetData.equity.push(currentEquity);
                    balanceSheetData.totalLiabilitiesAndEquity.push(totalLiabilities + currentEquity);

                    // --- Dynamic Profit Allocation ---
                    let allocatedNetProfit = netProfit; // Use net profit for allocation
                    let tier = profitAllocationTiers.find(t => allocatedNetProfit >= t.minRevenue && allocatedNetProfit <= t.maxRevenue);
                    if (!tier) tier = profitAllocationTiers[0]; // Default to first tier if not found (shouldn't happen with Infinity)

                    profitAllocationData.profitAccount.push(allocatedNetProfit * tier.allocations.profit);
                    profitAllocationData.ownerPay.push(allocatedNetProfit * tier.allocations.ownerPay);
                    profitAllocationData.taxAccount.push(allocatedNetProfit * tier.allocations.tax);
                    profitAllocationData.capExAccount.push(allocatedNetProfit * tier.allocations.capEx);
                    profitAllocationData.growthInvestmentAccount.push(allocatedNetProfit * tier.allocations.growthInvestment);
                }

                // --- Aggregate Monthly Data to Display Granularity ---
                const periods = [];
                const displayPnl = {};
                const displayCashFlow = {};
                const displayBalanceSheet = {};
                const displayKpis = {};
                const displayProfitAllocation = {};

                const aggregateData = (dataArray, aggregationType) => {
                    const aggregated = [];
                    let currentSum = 0;
                    let count = 0;
                    let divisor = 1; // For averages

                    if (aggregationType === 'quarterly') divisor = 3;
                    if (aggregationType === 'annual') divisor = 12;

                    for (let i = 0; i < dataArray.length; i++) {
                        currentSum += dataArray[i];
                        count++;
                        if (count === divisor || i === dataArray.length - 1) {
                            aggregated.push(currentSum);
                            currentSum = 0;
                            count = 0;
                        }
                    }
                    return aggregated;
                };

                const aggregateBalanceSheet = (dataArray, aggregationType) => {
                    const aggregated = [];
                    let divisor = 1; // For averages

                    if (aggregationType === 'quarterly') divisor = 3;
                    if (aggregationType === 'annual') divisor = 12;

                    for (let i = divisor - 1; i < dataArray.length; i += divisor) {
                        aggregated.push(dataArray[i]); // Take the value at the end of the period
                    }
                    return aggregated;
                };

                const aggregateKpi = (numeratorArray, denominatorArray, aggregationType) => {
                    const aggregated = [];
                    let numSum = 0;
                    let denSum = 0;
                    let count = 0;
                    let divisor = 1;

                    if (aggregationType === 'quarterly') divisor = 3;
                    if (aggregationType === 'annual') divisor = 12;

                    for (let i = 0; i < numeratorArray.length; i++) {
                        numSum += numeratorArray[i];
                        denSum += denominatorArray[i];
                        count++;
                        if (count === divisor || i === numeratorArray.length - 1) {
                            aggregated.push(denSum !== 0 ? numSum / denSum : 0);
                            numSum = 0;
                            denSum = 0;
                            count = 0;
                        }
                    }
                    return aggregated;
                };


                // Populate periods and aggregate data based on display granularity
                if (currentAssumptions.displayGranularity === 'monthly') {
                    for (let i = 0; i < totalMonths; i++) {
                        const date = new Date(2025, i, 1); // Start from Jan 2025
                        periods.push(date.toLocaleString('en-US', { month: 'short', year: 'numeric' }));
                    }
                    Object.keys(monthlyData).forEach(key => displayPnl[key] = monthlyData[key]);
                    Object.keys(cashFlowData).forEach(key => displayCashFlow[key] = cashFlowData[key]);
                    Object.keys(balanceSheetData).forEach(key => displayBalanceSheet[key] = balanceSheetData[key]);
                    Object.keys(profitAllocationData).forEach(key => displayProfitAllocation[key] = profitAllocationData[key]);

                    // KPIs for monthly
                    displayKpis.grossMargin = monthlyData.grossProfit.map((gp, i) => monthlyData.revenue[i] !== 0 ? gp / monthlyData.revenue[i] : 0);
                    displayKpis.netProfitMargin = monthlyData.netProfit.map((np, i) => monthlyData.revenue[i] !== 0 ? np / monthlyData.revenue[i] : 0);
                    displayKpis.debtToEquity = balanceSheetData.debt.map((debt, i) => balanceSheetData.equity[i] !== 0 ? debt / balanceSheetData.equity[i] : 0);
                    displayKpis.customerChurnRate = monthlyData.revenue.map((_, i) => currentAssumptions.churnRate); // Churn rate is an assumption
                    displayKpis.customerAcquisitionCost = monthlyData.revenue.map((_, i) => currentAssumptions.customerAcquisitionCost); // CAC is an assumption

                } else if (currentAssumptions.displayGranularity === 'quarterly') {
                    const totalQuarters = Math.ceil(totalMonths / 3);
                    for (let i = 0; i < totalQuarters; i++) {
                        periods.push(`Q${(i % 4) + 1} ${Math.floor(i / 4) + 1}`); // Q1 Y1, Q2 Y1, etc.
                    }
                    Object.keys(monthlyData).forEach(key => displayPnl[key] = aggregateData(monthlyData[key], 'quarterly'));
                    Object.keys(cashFlowData).forEach(key => displayCashFlow[key] = aggregateData(cashFlowData[key], 'quarterly'));
                    Object.keys(balanceSheetData).forEach(key => displayBalanceSheet[key] = aggregateBalanceSheet(balanceSheetData[key], 'quarterly'));
                    Object.keys(profitAllocationData).forEach(key => displayProfitAllocation[key] = aggregateData(profitAllocationData[key], 'quarterly'));

                    // KPIs for quarterly
                    displayKpis.grossMargin = aggregateKpi(monthlyData.grossProfit, monthlyData.revenue, 'quarterly');
                    displayKpis.netProfitMargin = aggregateKpi(monthlyData.netProfit, monthlyData.revenue, 'quarterly');
                    displayKpis.debtToEquity = aggregateBalanceSheet(balanceSheetData.debt, 'quarterly').map((debt, i) => {
                        const equity = aggregateBalanceSheet(balanceSheetData.equity, 'quarterly')[i];
                        return equity !== 0 ? debt / equity : 0;
                    });
                    displayKpis.customerChurnRate = displayKpis.grossMargin.map(() => currentAssumptions.churnRate);
                    displayKpis.customerAcquisitionCost = displayKpis.grossMargin.map(() => currentAssumptions.customerAcquisitionCost);


                } else if (currentAssumptions.displayGranularity === 'annual') {
                    const totalYears = currentAssumptions.timeHorizonYears;
                    for (let i = 0; i < totalYears; i++) {
                        periods.push(`Year ${i + 1}`);
                    }
                    Object.keys(monthlyData).forEach(key => displayPnl[key] = aggregateData(monthlyData[key], 'annual'));
                    Object.keys(cashFlowData).forEach(key => displayCashFlow[key] = aggregateData(cashFlowData[key], 'annual'));
                    Object.keys(balanceSheetData).forEach(key => displayBalanceSheet[key] = aggregateBalanceSheet(balanceSheetData[key], 'annual'));
                    Object.keys(profitAllocationData).forEach(key => displayProfitAllocation[key] = aggregateData(profitAllocationData[key], 'annual'));

                    // KPIs for annual
                    displayKpis.grossMargin = aggregateKpi(monthlyData.grossProfit, monthlyData.revenue, 'annual');
                    displayKpis.netProfitMargin = aggregateKpi(monthlyData.netProfit, monthlyData.revenue, 'annual');
                    displayKpis.debtToEquity = aggregateBalanceSheet(balanceSheetData.debt, 'annual').map((debt, i) => {
                        const equity = aggregateBalanceSheet(balanceSheetData.equity, 'annual')[i];
                        return equity !== 0 ? debt / equity : 0;
                    });
                    displayKpis.customerChurnRate = displayKpis.grossMargin.map(() => currentAssumptions.churnRate);
                    displayKpis.customerAcquisitionCost = displayKpis.grossMargin.map(() => currentAssumptions.customerAcquisitionCost);
                }

                // Store aggregated data for export/charts
                lastCalculatedData = {
                    periods,
                    pnl: displayPnl,
                    cashFlow: displayCashFlow,
                    balanceSheet: displayBalanceSheet,
                    kpis: displayKpis,
                    profitAllocation: displayProfitAllocation
                };


                // --- Render Tables ---
                function renderTable(tableBody, dataObject, formatFunc = formatCurrency) {
                    const headerRow = tableBody.previousElementSibling.querySelector('tr');
                    // Add period headers
                    periods.forEach(period => {
                        const th = document.createElement('th');
                        th.textContent = period;
                        headerRow.appendChild(th);
                    });

                    for (const key in dataObject) {
                        const row = document.createElement('tr');
                        const metricCell = document.createElement('td');
                        metricCell.textContent = formatMetricName(key);
                        row.appendChild(metricCell);

                        dataObject[key].forEach(value => {
                            const dataCell = document.createElement('td');
                            dataCell.textContent = isNaN(value) ? 'N/A' : formatFunc(value);
                            if (value < 0) {
                                dataCell.classList.add('negative'); // Add class for negative numbers
                            }
                            row.appendChild(dataCell);
                        });
                        tableBody.appendChild(row);
                    }
                }

                renderTable(pnlTableBody, {
                    revenue: displayPnl.revenue,
                    cogs: displayPnl.cogs,
                    grossProfit: displayPnl.grossProfit,
                    marketingExpenses: displayPnl.marketingExpenses,
                    salaryExpenses: displayPnl.salaryExpenses,
                    rentExpenses: displayPnl.rentExpenses,
                    otherOperatingExpenses: displayPnl.otherOperatingExpenses,
                    totalOperatingExpenses: displayPnl.totalOperatingExpenses,
                    depreciation: displayPnl.depreciation,
                    ebit: displayPnl.ebit,
                    interestExpense: displayPnl.interestExpense,
                    ebt: displayPnl.ebt,
                    taxes: displayPnl.taxes,
                    netProfit: displayPnl.netProfit
                });

                renderTable(cashFlowTableBody, {
                    netIncomeCF: displayCashFlow.netIncomeCF,
                    depreciationCF: displayCashFlow.depreciationCF,
                    changeInAR: displayCashFlow.changeInAR,
                    changeInAP: displayCashFlow.changeInAP,
                    netCashFromOperating: displayCashFlow.netCashFromOperating,
                    capexCF: displayCashFlow.capexCF,
                    netCashFromInvesting: displayCashFlow.netCashFromInvesting,
                    netChangeInCash: displayCashFlow.netChangeInCash,
                    beginningCash: displayCashFlow.beginningCash.slice(1), // Exclude initial cash balance
                    endingCash: displayCashFlow.endingCash.slice(1) // Exclude initial cash balance
                });

                renderTable(balanceSheetTableBody, {
                    cash: displayBalanceSheet.cash.slice(1),
                    accountsReceivable: displayBalanceSheet.accountsReceivable.slice(1),
                    ppe: displayBalanceSheet.ppe.slice(1),
                    totalAssets: displayBalanceSheet.totalAssets.slice(1),
                    accountsPayable: displayBalanceSheet.accountsPayable.slice(1),
                    debt: displayBalanceSheet.debt.slice(1),
                    totalLiabilities: displayBalanceSheet.totalLiabilities.slice(1),
                    equity: displayBalanceSheet.equity.slice(1),
                    totalLiabilitiesAndEquity: displayBalanceSheet.totalLiabilitiesAndEquity.slice(1)
                });

                renderTable(kpiTableBody, {
                    grossMargin: displayKpis.grossMargin,
                    netProfitMargin: displayKpis.netProfitMargin,
                    debtToEquity: displayKpis.debtToEquity,
                    customerChurnRate: displayKpis.customerChurnRate,
                    customerAcquisitionCost: displayKpis.customerAcquisitionCost
                }, (value) => {
                    // Custom formatting for KPIs
                    // The `key` parameter is not directly available here in this scope,
                    // so we need to pass the actual key or check the value type if needed.
                    // For now, assuming formatPercentage is for ratios and formatCurrency for costs.
                    if (value === currentAssumptions.customerAcquisitionCost) { // Check if it's the CAC value
                        return formatCurrency(value);
                    }
                    return formatPercentage(value);
                });

                renderTable(profitAllocationTableBody, {
                    profitAccount: profitAllocationData.profitAccount,
                    ownerPay: profitAllocationData.ownerPay,
                    taxAccount: profitAllocationData.taxAccount,
                    capExAccount: profitAllocationData.capExAccount,
                    growthInvestmentAccount: profitAllocationData.growthInvestmentAccount
                });

                // Render charts after tables
                renderCharts(periods, lastCalculatedData);
                saveInputsToLocalStorage(); // Save inputs after successful calculation
            }

            function downloadCSV() {
                if (!lastCalculatedData.periods || lastCalculatedData.periods.length === 0) {
                    alert("Please generate the financial model first.");
                    return;
                }

                const header = ['Metric', ...lastCalculatedData.periods];
                let csvContent = header.map(h => `"${h}"`).join(',') + '\n';

                const addRow = (metricName, dataArray, formatFunc = formatCurrency) => {
                    const formattedData = dataArray.map(val => isNaN(val) ? 'N/A' : formatFunc(val).replace('$', '').replace('%', '')); // Remove currency/percent signs for CSV
                    csvContent += `"${metricName}",${formattedData.map(d => `"${d}"`).join(',')}\n`;
                };

                // P&L Data
                addRow('Revenue', lastCalculatedData.pnl.revenue);
                addRow('COGS', lastCalculatedData.pnl.cogs);
                addRow('Gross Profit', lastCalculatedData.pnl.grossProfit);
                addRow('Marketing Expenses', lastCalculatedData.pnl.marketingExpenses);
                addRow('Salary Expenses', lastCalculatedData.pnl.salaryExpenses);
                addRow('Rent Expenses', lastCalculatedData.pnl.rentExpenses);
                addRow('Other Operating Expenses', lastCalculatedData.pnl.otherOperatingExpenses);
                addRow('Total Operating Expenses', lastCalculatedData.pnl.totalOperatingExpenses);
                addRow('Depreciation', lastCalculatedData.pnl.depreciation);
                addRow('EBIT', lastCalculatedData.pnl.ebit);
                addRow('Interest Expense', lastCalculatedData.pnl.interestExpense);
                addRow('EBT', lastCalculatedData.pnl.ebt);
                addRow('Taxes', lastCalculatedData.pnl.taxes);
                addRow('Net Profit', lastCalculatedData.pnl.netProfit);

                // Cash Flow Data (slice to match periods)
                addRow('Net Income (CF)', lastCalculatedData.cashFlow.netIncomeCF);
                addRow('Depreciation (CF)', lastCalculatedData.cashFlow.depreciationCF);
                addRow('Net Cash From Operating', lastCalculatedData.cashFlow.netCashFromOperating);
                addRow('CapEx (CF)', lastCalculatedData.cashFlow.capexCF);
                addRow('Net Cash From Investing', lastCalculatedData.cashFlow.netCashFromInvesting);
                addRow('Net Change In Cash', lastCalculatedData.cashFlow.netChangeInCash);
                addRow('Beginning Cash', lastCalculatedData.cashFlow.beginningCash.slice(1));
                addRow('Ending Cash', lastCalculatedData.cashFlow.endingCash.slice(1));

                // Balance Sheet Data (slice to match periods)
                addRow('Cash (BS)', lastCalculatedData.balanceSheet.cash.slice(1));
                addRow('PPE (Net)', lastCalculatedData.balanceSheet.ppe.slice(1));
                addRow('Total Assets', lastCalculatedData.balanceSheet.totalAssets.slice(1));
                addRow('Debt', lastCalculatedData.balanceSheet.debt.slice(1));
                addRow('Total Liabilities', lastCalculatedData.balanceSheet.totalLiabilities.slice(1));
                addRow('Equity', lastCalculatedData.balanceSheet.equity.slice(1));
                addRow('Total Liabilities & Equity', lastCalculatedData.balanceSheet.totalLiabilitiesAndEquity.slice(1));


                // KPIs
                addRow('Gross Margin', lastCalculatedData.kpis.grossMargin, formatPercentage);
                addRow('Net Profit Margin', lastCalculatedData.kpis.netProfitMargin, formatPercentage);
                addRow('Debt To Equity', lastCalculatedData.kpis.debtToEquity, (val) => isNaN(val) ? 'N/A' : val.toFixed(2));
                addRow('Customer Churn Rate', lastCalculatedData.kpis.customerChurnRate, formatPercentage);
                addRow('Customer Acquisition Cost', lastCalculatedData.kpis.customerAcquisitionCost, formatCurrency);

                // Profit Allocation
                addRow('Profit Account', lastCalculatedData.profitAllocation.profitAccount);
                addRow('Owner Pay', lastCalculatedData.profitAllocation.ownerPay);
                addRow('Tax Account', lastCalculatedData.profitAllocation.taxAccount);
                addRow('CapEx Allocation', lastCalculatedData.profitAllocation.capExAccount);
                addRow('Growth Investment', lastCalculatedData.profitAllocation.growthInvestmentAccount);


                const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.setAttribute('href', url);
                link.setAttribute('download', 'financial_model_data.csv');
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(url);
            }

                const { jsPDF } = window.jspdf;

                    doc.text(title, 14, currentY);

                        didDrawPage: function (data) {
                            // Footer for page numbers
                            let str = "Page " + doc.internal.getNumberOfPages();
                            doc.setFontSize(8);
                            doc.text(str, data.settings.margin.left, doc.internal.pageSize.height - 10);

                    currentY = doc.autoTable.previous.finalY + 10; // Update Y position for next table

                // Add P&L

                    cogs: lastCalculatedData.pnl.cogs,
                    grossProfit: lastCalculatedData.pnl.grossProfit,
                    marketingExpenses: lastCalculatedData.pnl.marketingExpenses,
                    salaryExpenses: lastCalculatedData.pnl.salaryExpenses,
                    rentExpenses: lastCalculatedData.pnl.rentExpenses,
                    otherOperatingExpenses: lastCalculatedData.pnl.otherOperatingExpenses,
                    totalOperatingExpenses: lastCalculatedData.pnl.totalOperatingExpenses,
                    depreciation: lastCalculatedData.pnl.depreciation,
                    ebit: lastCalculatedData.pnl.ebit,
                    interestExpense: lastCalculatedData.pnl.interestExpense,
                    ebt: lastCalculatedData.pnl.ebt,
                    taxes: lastCalculatedData.pnl.taxes,
                    netProfit: lastCalculatedData.pnl.netProfit
                });

                // Add Cash Flow
                addTableToPDF('Cash Flow Statement', {
                    netIncomeCF: lastCalculatedData.cashFlow.netIncomeCF,
                    depreciationCF: lastCalculatedData.cashFlow.depreciationCF,
                    netCashFromOperating: lastCalculatedData.cashFlow.netCashFromOperating,
                    capexCF: lastCalculatedData.cashFlow.capexCF,
                    netCashFromInvesting: lastCalculatedData.cashFlow.netCashFromInvesting,
                    netChangeInCash: lastCalculatedData.cashFlow.netChangeInCash,
                    beginningCash: lastCalculatedData.cashFlow.beginningCash.slice(1),
                    endingCash: lastCalculatedData.cashFlow.endingCash.slice(1)
                });

                // Add Balance Sheet
                addTableToPDF('Balance Sheet', {
                    cash: lastCalculatedData.balanceSheet.cash.slice(1),
                    ppe: lastCalculatedData.balanceSheet.ppe.slice(1),
                    totalAssets: lastCalculatedData.balanceSheet.totalAssets.slice(1),
                    debt: lastCalculatedData.balanceSheet.debt.slice(1),
                    totalLiabilities: lastCalculatedData.balanceSheet.totalLiabilities.slice(1),
                    equity: lastCalculatedData.balanceSheet.equity.slice(1),
                    totalLiabilitiesAndEquity: lastCalculatedData.balanceSheet.totalLiabilitiesAndEquity.slice(1)
                });

                // Add KPIs
                addTableToPDF('Key Performance Indicators (KPIs)', {
                    grossMargin: lastCalculatedData.kpis.grossMargin,
                    netProfitMargin: lastCalculatedData.kpis.netProfitMargin,
                    debtToEquity: lastCalculatedData.kpis.debtToEquity,
                    customerChurnRate: lastCalculatedData.kpis.customerChurnRate,
                    customerAcquisitionCost: lastCalculatedData.kpis.customerAcquisitionCost
                }, (value, key) => {
                    // This function is called with value and the *index* of the value in the array, not the key name.
                    // To correctly format based on metric name for PDF, we need a different approach or to pass the key explicitly.
                    // For now, let's assume if the value is a direct match for a known currency assumption, format as currency.
                    // Otherwise, format as percentage. This is a heuristic.
                    if (value === lastCalculatedData.kpis.customerAcquisitionCost[0] && lastCalculatedData.kpis.customerAcquisitionCost.length > 0) {
                        return formatCurrency(value);
                    }
                    return formatPercentage(value);
                });

                // Add Profit Allocation
                addTableToPDF('Dynamic Profit Allocation', {
                    profitAccount: lastCalculatedData.profitAllocation.profitAccount,
                    ownerPay: lastCalculatedData.profitAllocation.ownerPay,
                    taxAccount: lastCalculatedData.profitAllocation.taxAccount,
                    capExAccount: lastCalculatedData.profitAllocation.capExAccount,
                    growthInvestmentAccount: lastCalculatedData.profitAllocation.growthInvestmentAccount
                });


                // Add Charts as images to PDF
                doc.addPage();
                currentY = 20;
                doc.setFontSize(14);
                doc.text("Financial Charts", 14, currentY);
                currentY += 10;

                const chartCanvases = document.querySelectorAll('#charts canvas');
                for (const canvas of chartCanvases) {
                    const imgData = canvas.toDataURL('image/png');
                    const imgWidth = 250; // Adjust width to fit landscape A4
                    const imgHeight = (canvas.height * imgWidth) / canvas.width;

                    if (currentY + imgHeight > doc.internal.pageSize.height - 20) {
                        doc.addPage();
                        currentY = 20;
                    }
                    doc.addImage(imgData, 'PNG', 14, currentY, imgWidth, imgHeight);
                    currentY += imgHeight + 10;
                }

                doc.save('financial_model.pdf');
            }

            calculateModelBtn.addEventListener('click', calculateAndDisplayModel);
            applyScenarioBtn.addEventListener('click', () => {
                applyScenario(scenarioSelect.value);
            });
            downloadCSVBtn.addEventListener('click', downloadCSV);
            exportPDFBtn.addEventListener('click', exportPDF);


            // Tab switching logic
            tabButtons.forEach(button => {
                button.addEventListener('click', (event) => {
                    // Deactivate all tabs and hide all content
                    tabButtons.forEach(btn => btn.classList.remove('active'));
                    tabContents.forEach(content => content.classList.remove('active'));

                    // Activate clicked tab and show its content
                    event.currentTarget.classList.add('active');
                    const targetTabId = event.currentTarget.dataset.tab;
                    document.getElementById(targetTabId).classList.add('active');
                });
            });

            // Tooltip hover logic
            infoIcons.forEach(icon => {
                icon.addEventListener('mouseover', () => {
                    const tooltipId = icon.dataset.tooltipId;
                    const tooltipElement = document.getElementById(`tooltip-${tooltipId}`);
                    if (tooltipElement && tooltipDefinitions[tooltipId]) {
                        tooltipElement.textContent = tooltipDefinitions[tooltipId];
                        tooltipElement.style.visibility = 'visible';
                        tooltipElement.style.opacity = '1';
                    }
                });

                icon.addEventListener('mouseout', () => {
                    const tooltipId = icon.dataset.tooltipId;
                    const tooltipElement = document.getElementById(`tooltip-${tooltipId}`);
                    if (tooltipElement) {
                        tooltipElement.style.visibility = 'hidden';
                        tooltipElement.style.opacity = '0';
                    }
                });
            });

            // Initial load: Load saved inputs, apply default scenario if none, and calculate
            loadInputsFromLocalStorage();
            document.querySelector('.tab-button').click(); // Activate first tab on load
        });
