## Crypto Tax Calculations

### Capital Gains Calculator

```javascript
function calculateCryptoGains(transactions, method = 'FIFO') {
  const holdings = {}
  const gains = {
    shortTerm: [],  // < 1 year
    longTerm: []    // >= 1 year
  }
  
  transactions.forEach(tx => {
    if (tx.type === 'buy') {
      // Add to holdings with cost basis
      holdings[tx.crypto] = holdings[tx.crypto] || []
      holdings[tx.crypto].push({
        amount: tx.amount,
        costBasis: tx.pricePerCoin,
        date: tx.date
      })
    } else if (tx.type === 'sell') {
      // Match with purchases using FIFO/LIFO
      const proceeds = tx.amount * tx.pricePerCoin
      const { costBasis, purchaseDate } = matchTaxLot(holdings[tx.crypto], tx.amount, method)
      
      const gain = proceeds - costBasis
      const holdingPeriod = daysBetween(purchaseDate, tx.date)
      
      if (holdingPeriod < 365) {
        gains.shortTerm.push({ gain, tx })
      } else {
        gains.longTerm.push({ gain, tx })
      }
    }
  })
  
  return gains
}
```

### Income Calculation

```javascript
function calculateCryptoIncome(transactions) {
  const income = {
    mining: 0,
    staking: 0,
    airdrops: 0,
    interest: 0,
    other: 0
  }
  
  transactions.forEach(tx => {
    if (['mining', 'staking_reward', 'airdrop', 'interest'].includes(tx.type)) {
      const incomeValue = tx.amount * tx.fairMarketValue
      income[tx.type] += incomeValue
    }
  })
  
  return income
}
```
