### File Validation

#### Client-Side Validation

```javascript
const FILE_VALIDATION = {
  maxSize: 10 * 1024 * 1024, // 10MB
  maxFiles: 20, // Per upload batch
  allowedTypes: [
    'application/pdf',
    'image/jpeg',
    'image/png',
    'image/heic',
    'application/msword',
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'application/vnd.ms-excel',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    'text/csv'
  ],
  allowedExtensions: ['.pdf', '.jpg', '.jpeg', '.png', '.heic', '.doc', '.docx', '.xlsx', '.xls', '.csv']
}

function validateFile(file) {
  const errors = []
  
  // Check file size
  if (file.size > FILE_VALIDATION.maxSize) {
    errors.push(`File too large. Maximum size is ${formatFileSize(FILE_VALIDATION.maxSize)}`)
  }
  
  // Check file type
  if (!FILE_VALIDATION.allowedTypes.includes(file.type)) {
    const extension = file.name.split('.').pop()
    if (!FILE_VALIDATION.allowedExtensions.includes(`.${extension}`)) {
      errors.push(`File type not supported. Allowed types: ${FILE_VALIDATION.allowedExtensions.join(', ')}`)
    }
  }
  
  // Check file name
  if (file.name.length > 255) {
    errors.push('File name too long')
  }
  
  return { isValid: errors.length === 0, errors }
}
```

#### Image Optimization (Before Upload)

```javascript
// Compress large images before uploading
async function compressImage(file, maxWidth = 1920, quality = 0.8) {
  return new Promise((resolve) => {
    const reader = new FileReader()
    reader.onload = (e) => {
      const img = new Image()
      img.onload = () => {
        const canvas = document.createElement('canvas')
        let { width, height } = img
        
        if (width > maxWidth) {
          height = (height * maxWidth) / width
          width = maxWidth
        }
        
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        ctx.drawImage(img, 0, 0, width, height)
        
        canvas.toBlob(resolve, 'image/jpeg', quality)
      }
      img.src = e.target.result
    }
    reader.readAsDataURL(file)
  })
}
```

### Upload API Implementation

#### Chunked Upload (for large files)

```javascript
async function uploadFileInChunks(file, onProgress) {
  const CHUNK_SIZE = 1024 * 1024 // 1MB chunks
  const totalChunks = Math.ceil(file.size / CHUNK_SIZE)
  const uploadId = generateUploadId()
  
  for (let chunkIndex = 0; chunkIndex < totalChunks; chunkIndex++) {
    const start = chunkIndex * CHUNK_SIZE
    const end = Math.min(start + CHUNK_SIZE, file.size)
    const chunk = file.slice(start, end)
    
    const formData = new FormData()
    formData.append('chunk', chunk)
    formData.append('chunkIndex', chunkIndex)
    formData.append('totalChunks', totalChunks)
    formData.append('uploadId', uploadId)
    formData.append('fileName', file.name)
    
    await fetch('/api/documents/upload-chunk', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${token}`
      },
      body: formData
    })
    
    onProgress((chunkIndex + 1) / totalChunks * 100)
  }
  
  // Finalize upload
  return await finalizeUpload(uploadId)
}
```

```javascript
// Express.js endpoint
const multer = require('multer')
const path = require('path')
const crypto = require('crypto')

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    const userId = req.user.id
    const uploadPath = path.join(__dirname, '../uploads', userId)
    fs.mkdirSync(uploadPath, { recursive: true })
    cb(null, uploadPath)
  },
  filename: (req, file, cb) => {
    const uniqueName = `${Date.now()}-${crypto.randomBytes(6).toString('hex')}${path.extname(file.originalname)}`
    cb(null, uniqueName)
  }
})

const upload = multer({
  storage,
  limits: { fileSize: 10 * 1024 * 1024 }, // 10MB
  fileFilter: (req, file, cb) => {
    const allowedTypes = /pdf|jpg|jpeg|png|doc|docx|xlsx|csv/
    const extname = allowedTypes.test(path.extname(file.originalname).toLowerCase())
    const mimetype = allowedTypes.test(file.mimetype)
    
    if (extname && mimetype) {
      cb(null, true)
    } else {
      cb(new Error('Invalid file type'))
    }
  }
})

router.post('/documents/upload', upload.array('documents', 20), async (req, res) => {
  try {
    const { category } = req.body
    const userId = req.user.id
    
    // Process uploaded files
    const documents = await Promise.all(
      req.files.map(async (file) => {
        // Run virus scan
        await scanForViruses(file.path)
        
        // Run OCR if PDF or image
        let ocrText = null
        let suggestedCategory = category
        
        if (['.pdf', '.jpg', '.jpeg', '.png'].includes(path.extname(file.originalname).toLowerCase())) {
          ocrText = await performOCR(file.path)

          userId,

          ocrText,
          uploadDate: new Date(),

        })
        
        return document
      })
    )
    
    res.json({ success: true, documents })
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})
```

```javascript
const Tesseract = require('tesseract.js')

async function performOCR(imagePath) {
  const { data: { text } } = await Tesseract.recognize(
    imagePath,
    'eng',
    {

    }
```

### Document Classification

#### AI-Powered Classification

```javascript
// Use extracted OCR text to classify document type
async function classifyDocument(ocrText) {
  const classifications = {
    'W2': /W-2|Wage and Tax Statement|Employer Identification Number/i,
    '1099-NEC': /1099-NEC|Nonemployee Compensation/i,
    '1099-INT': /1099-INT|Interest Income/i,
    '1099-DIV': /1099-DIV|Dividends and Distributions/i,
    '1099-B': /1099-B|Proceeds from Broker/i,
    '1099-K': /1099-K|Payment Card and Third Party/i,
    '1098': /1098|Mortgage Interest Statement/i,
    'Receipt': /receipt|invoice|paid|total|subtotal/i,
    'Bank Statement': /bank statement|account summary|beginning balance|ending balance/i,
    'Crypto Transaction': /bitcoin|ethereum|cryptocurrency|wallet|blockchain/i,
    'Business Expense': /expense|vendor|supplier|purchase order/i
  }
  
  for (const [category, pattern] of Object.entries(classifications)) {
    if (pattern.test(ocrText)) {
      return category
    }
  }
  
  return 'Other Documents'
}
```

#### Extract Key Information

```javascript
// Extract specific data points from tax forms
function extractW2Data(ocrText) {
  const w2Data = {
    employerName: extractPattern(ocrText, /employer[^\n]*\n([^\n]+)/i),
    employerEIN: extractPattern(ocrText, /EIN[:\s]+(\d{2}-\d{7})/i),
    wages: extractPattern(ocrText, /wages.*?(\d+[\d,]*\.?\d{0,2})/i),
    federalWithholding: extractPattern(ocrText, /federal.*withheld.*?(\d+[\d,]*\.?\d{0,2})/i),
    socialSecurityWages: extractPattern(ocrText, /social security wages.*?(\d+[\d,]*\.?\d{0,2})/i),
    medicareWages: extractPattern(ocrText, /medicare wages.*?(\d+[\d,]*\.?\d{0,2})/i)
  }
  
  return w2Data
}

function extractPattern(text, regex) {
  const match = text.match(regex)
  return match ? match[1].trim() : null
}
```

### Quality Check & Validation

```javascript
// Check if scanned document is readable
async function assessDocumentQuality(imagePath) {
  const metadata = await sharp(imagePath).metadata()
  const stats = await sharp(imagePath).stats()
  
  const qualityIssues = []
  
  // Check resolution
  if (metadata.density < 150) {
    qualityIssues.push('Low resolution. Please rescan at higher quality.')
  }
  
  // Check if too dark/light
  const brightness = (stats.channels[0].mean + stats.channels[1].mean + stats.channels[2].mean) / 3
  if (brightness < 50) {
    qualityIssues.push('Image too dark. Please use better lighting.')
  } else if (brightness > 200) {
    qualityIssues.push('Image too bright. May be overexposed.')
  }
  
  // Check if blurry (using Laplacian variance)
  // Lower variance = more blurry
  const blurriness = await detectBlurriness(imagePath)
  if (blurriness < 100) {
    qualityIssues.push('Image appears blurry. Please take a sharper photo.')
  }
  
  return {
    isGoodQuality: qualityIssues.length === 0,
    issues: qualityIssues
  }
}
```

```javascript
const sharp = require('sharp')

async function generateThumbnail(filePath, outputPath) {
  if (filePath.endsWith('.pdf')) {
    // Convert first page of PDF to image thumbnail
    const { PDFImage } = require('pdf-image')
    const pdfImage = new PDFImage(filePath)
    const imagePath = await pdfImage.convertPage(0)
    
    await sharp(imagePath)

    // Generate thumbnail from image

  }
}
```

```javascript
const DOCUMENT_STATUSES = {

function updateDocumentStatus(documentId, newStatus, adminNotes) {
  // Update database
  await db.documents.update({
    where: { id: documentId },
    data: { 
      status: newStatus,
      adminNotes,
      reviewedAt: new Date()
    }

      to: document.user.email,
      subject: 'Additional Information Needed',
      template: 'document-needs-info',
      data: { documentName: document.fileName, notes: adminNotes }

  }
```

## Security & Compliance

### Virus Scanning

```javascript
const ClamScan = require('clamscan')

async function scanForViruses(filePath) {
  const clamscan = await new ClamScan().init({
    clamdscan: {
      host: 'localhost',
      port: 3310
    }
  })
  
  const { isInfected, viruses } = await clamscan.isInfected(filePath)
  
  if (isInfected) {
    fs.unlinkSync(filePath) // Delete infected file
    throw new Error(`File infected with: ${viruses.join(', ')}`)
  }
}
```

### Encryption at Rest

```javascript
const crypto = require('crypto')

function encryptFile(inputPath, outputPath, key) {
  const cipher = crypto.createCipher('aes-256-cbc', key)
  const input = fs.createReadStream(inputPath)
  const output = fs.createWriteStream(outputPath)
  
  input.pipe(cipher).pipe(output)
}

function decryptFile(inputPath, outputPath, key) {
  const decipher = crypto.createDecipher('aes-256-cbc', key)
  const input = fs.createReadStream(inputPath)
  const output = fs.createWriteStream(outputPath)
  
  input.pipe(decipher).pipe(output)
}
```
