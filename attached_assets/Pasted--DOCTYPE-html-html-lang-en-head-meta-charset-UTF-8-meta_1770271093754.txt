    <script>
        let items = JSON.parse(localStorage.getItem('pl_items')) || [];
        let barChart, pieChart;

        function formatCurrency(n) {
            return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n);
        }

        function clearData() {
            if(confirm("Are you sure you want to clear this weekly report?")) {
                items = [];
                localStorage.removeItem('pl_items');
                updateCalculations();
            }
        }

        function removeItem(index) {
            items.splice(index, 1);
            localStorage.setItem('pl_items', JSON.stringify(items));
            updateCalculations();
        }

        document.getElementById('item-form').onsubmit = (e) => {
            e.preventDefault();
            const newItem = {
                type: document.getElementById('type').value,
                amount: parseFloat(document.getElementById('amount').value),
                desc: document.getElementById('desc').value,
                date: new Date().toLocaleDateString()
            };
            items.unshift(newItem);
            localStorage.setItem('pl_items', JSON.stringify(items));
            e.target.reset();
            updateCalculations();
        };

        function updateCalculations() {
            let rev = 0, cogs = 0, opex = 0;
            const historyList = document.getElementById('history-list');
            historyList.innerHTML = '';

            items.forEach((item, idx) => {
                if(item.type === 'revenue') rev += item.amount;
                if(item.type === 'cogs') cogs += item.amount;
                if(item.type === 'opex') opex += item.amount;

                // Render History Item
                const div = document.createElement('div');
                div.className = "flex justify-between items-center p-3 bg-white/50 rounded-lg border border-white/40 line-item transition";
                const colorClass = item.type === 'revenue' ? 'text-emerald-600' : 'text-slate-600';
                div.innerHTML = `
                    <div class="flex flex-col">
                        <span class="text-[10px] font-black uppercase text-slate-400">${item.type}</span>
                        <span class="text-sm font-bold text-slate-700">${item.desc}</span>
                    </div>
                    <div class="text-right">
                        <p class="text-sm font-bold ${colorClass}">${item.type !== 'revenue' ? '-' : ''}${formatCurrency(item.amount)}</p>
                        <button onclick="removeItem(${idx})" class="text-[10px] text-red-400 hover:text-red-600 font-bold uppercase tracking-tighter">Delete</button>
                    </div>
                `;
                historyList.appendChild(div);
            });

            // Math Logic
            const gp = rev - cogs;
            const net = gp - opex;
            const taxRate = parseFloat(document.getElementById('tax-rate-input').value) / 100;
            const tax = net > 0 ? net * taxRate : 0;
            const takeHome = net - tax;

            // Update UI
            document.getElementById('stat-revenue').innerText = formatCurrency(rev);
            document.getElementById('stat-gross-profit').innerText = formatCurrency(gp);
            document.getElementById('stat-net-income').innerText = formatCurrency(net);
            document.getElementById('stat-take-home').innerText = formatCurrency(takeHome);

            document.getElementById('pl-revenue').innerText = formatCurrency(rev);
            document.getElementById('pl-cogs').innerText = `(${formatCurrency(cogs)})`;
            document.getElementById('pl-gross-profit').innerText = formatCurrency(gp);
            document.getElementById('pl-opex').innerText = `(${formatCurrency(opex)})`;
            document.getElementById('pl-net-income').innerText = formatCurrency(net);
            document.getElementById('pl-tax').innerText = `(${formatCurrency(tax)})`;
            document.getElementById('pl-take-home').innerText = formatCurrency(takeHome);

            renderCharts(rev, cogs, opex, net);
        }

        function renderCharts(rev, cogs, opex, net) {
            const barCtx = document.getElementById('barChart').getContext('2d');
            const pieCtx = document.getElementById('pieChart').getContext('2d');

            if(barChart) barChart.destroy();
            if(pieChart) pieChart.destroy();

            barChart = new Chart(barCtx, {
                type: 'bar',
                data: {
                    labels: ['Revenue', 'Expenses', 'Net'],
                    datasets: [{
                        data: [rev, (cogs + opex), net],
                        backgroundColor: ['#10b981', '#f43f5e', '#3b82f6'],
                        borderRadius: 8
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: { y: { beginAtZero: true, grid: { display: false } }, x: { grid: { display: false } } }
                }
            });

            pieChart = new Chart(pieCtx, {
                type: 'doughnut',
                data: {
                    labels: ['COGS', 'OPEX', 'Profit'],
                    datasets: [{
                        data: [cogs, opex, net > 0 ? net : 0],
                        backgroundColor: ['#f59e0b', '#ef4444', '#10b981'],
                        borderWidth: 0,
                        hoverOffset: 10
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { position: 'bottom' } }
                }
            });
        }

        window.onload = updateCalculations;
    </script>
