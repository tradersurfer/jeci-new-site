## Architecture Options

### Option 1: WebSocket (Real-Time)

**Best for:** True real-time communication, instant delivery

```javascript
// Using Socket.io
const io = require('socket.io')(server)

io.on('connection', (socket) => {
  console.log('User connected:', socket.id)
  
  // Join user's personal room
  socket.on('join', (userId) => {
    socket.join(`user_${userId}`)
  })
  
  // Handle incoming messages
  socket.on('send_message', async (data) => {
    const message = await saveMessage(data)
    
    // Emit to recipient
    io.to(`user_${data.recipientId}`).emit('new_message', message)
    
    // Send push notification if offline
    if (!isUserOnline(data.recipientId)) {
      await sendPushNotification(data.recipientId, message)
    }
  })
  
  // Typing indicator
  socket.on('typing', (data) => {
    io.to(`user_${data.recipientId}`).emit('user_typing', {
      userId: data.userId,
      isTyping: true
    })
  })
  
  socket.on('disconnect', () => {
    console.log('User disconnected:', socket.id)
  })
})
```

### Option 2: Polling (Simpler)

**Best for:** Easier implementation, works with any hosting

```javascript
// Poll for new messages every 3 seconds
useEffect(() => {
  const interval = setInterval(async () => {
    const response = await fetch('/api/messages/new', {
      headers: { 'Authorization': `Bearer ${token}` }
    })
    const newMessages = await response.json()
    
    if (newMessages.length > 0) {
      setMessages(prev => [...prev, ...newMessages])
      playNotificationSound()
    }
  }, 3000)
  
  return () => clearInterval(interval)
}, [])
```

```jsx
function ChatInterface() {
  const [messages, setMessages] = useState([])
  const [newMessage, setNewMessage] = useState('')
  const [isTyping, setIsTyping] = useState(false)
  const [attachments, setAttachments] = useState([])
  const messagesEndRef = useRef(null)
  const fileInputRef = useRef(null)
  
  useEffect(() => {
    loadMessageHistory()
    initializeWebSocket()
    scrollToBottom()
  }, [])
  
  useEffect(() => {
    scrollToBottom()
  }, [messages])
  
  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
  }
  
  const loadMessageHistory = async () => {
    const response = await fetch('/api/messages', {
      headers: { 'Authorization': `Bearer ${token}` }
    })
    const data = await response.json()
    setMessages(data.messages)
  }
  
  const handleSendMessage = async () => {
    if (!newMessage.trim() && attachments.length === 0) return
    
    const messageData = {
      text: newMessage.trim(),
      attachments: attachments.map(a => a.id),
      recipientId: getTaxProId(),
      sentAt: new Date().toISOString()
    }
    
    // Optimistic UI update
    const tempMessage = {
      ...messageData,
      id: `temp_${Date.now()}`,
      status: 'sending'
    }
    setMessages(prev => [...prev, tempMessage])
    
    try {
      const response = await fetch('/api/messages/send', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify(messageData)
      })
      
      const sentMessage = await response.json()
      
      // Replace temp message with real one
      setMessages(prev => 
        prev.map(m => m.id === tempMessage.id ? sentMessage : m)
      )
      
      // Clear input
      setNewMessage('')
      setAttachments([])
    } catch (error) {
      // Mark message as failed
      setMessages(prev => 
        prev.map(m => m.id === tempMessage.id ? {...m, status: 'failed'} : m)
      )
    }
  }
  
  const handleTyping = () => {
    if (!isTyping) {
      socket.emit('typing', {
        userId: currentUserId,
        recipientId: getTaxProId()
      })
      setIsTyping(true)
      
      setTimeout(() => setIsTyping(false), 3000)
    }
  }
  
  const handleFileAttachment = async (files) => {
    const uploadedFiles = await Promise.all(
      Array.from(files).map(file => uploadAttachment(file))
    )
    setAttachments(prev => [...prev, ...uploadedFiles])
  }
  
  return (
    <ChatContainer>
      <ChatHeader>
        <TaxProAvatar src={taxPro.avatar} />
        <TaxProInfo>
          <h3>{taxPro.name}</h3>
          <StatusIndicator online={taxPro.isOnline}>
            {taxPro.isOnline ? 'Online' : 'Offline'}
          </StatusIndicator>
        </TaxProInfo>
      </ChatHeader>
      
      <MessagesContainer>
        {messages.map(message => (
          <MessageBubble 
            key={message.id}
            message={message}
            isOwn={message.senderId === currentUserId}
          />
        ))}
        
        {isTyping && (
          <TypingIndicator>
            <span>{taxPro.name} is typing</span>
            <TypingDots />
          </TypingIndicator>
        )}
        
        <div ref={messagesEndRef} />
      </MessagesContainer>
      
      {attachments.length > 0 && (
        <AttachmentPreview>
          {attachments.map(file => (
            <AttachmentChip key={file.id}>
              <FileIcon type={file.type} />
              <span>{file.name}</span>
              <RemoveButton onClick={() => removeAttachment(file.id)}>
                ×
              </RemoveButton>
            </AttachmentChip>
          ))}
        </AttachmentPreview>
      )}
      
      <MessageInputContainer>
        <input
          type="file"
          ref={fileInputRef}
          style={{ display: 'none' }}
          multiple
          onChange={(e) => handleFileAttachment(e.target.files)}
        />
        
        <AttachButton onClick={() => fileInputRef.current.click()}>
          📎
        </AttachButton>
        
        <MessageInput
          value={newMessage}
          onChange={(e) => {
            setNewMessage(e.target.value)
            handleTyping()
          }}
          onKeyPress={(e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault()
              handleSendMessage()
            }
          }}
          placeholder="Type your message..."
          multiline
        />
        
        <SendButton 
          onClick={handleSendMessage}
          disabled={!newMessage.trim() && attachments.length === 0}
        >
          Send
        </SendButton>
      </MessageInputContainer>
    </ChatContainer>
  )
}
```

```javascript
const express = require('express')
const router = express.Router()

// Get conversation history
router.get('/messages', authenticateUser, async (req, res) => {
  const userId = req.user.id
  const { limit = 50, before } = req.query
  
  try {
    const messages = await db.messages.findMany({
      where: {
        OR: [
          { senderId: userId },
          { recipientId: userId }
        ],
        ...(before && { createdAt: { lt: new Date(before) } })
      },
      include: {
        attachments: true,
        sender: {
          select: { id: true, fullName: true, avatar: true }
        }
      },
      orderBy: { createdAt: 'desc' },
      take: limit
    })
    
    res.json({ messages: messages.reverse() })
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})

// Send new message
router.post('/messages/send', authenticateUser, async (req, res) => {
  const senderId = req.user.id
  const { text, recipientId, attachments } = req.body
  
  try {
    // Validate recipient exists and user has permission to message them
    const recipient = await db.users.findUnique({
      where: { id: recipientId }
    })
    
    if (!recipient) {
      return res.status(404).json({ error: 'Recipient not found' })
    }
    
    // Create message
    const message = await db.messages.create({
      data: {
        senderId,
        recipientId,
        text: text || null,
        status: 'sent',
        sentAt: new Date()
      },
      include: {
        sender: {
          select: { id: true, fullName: true, avatar: true }
        }
      }
    })
    
    // Link attachments
    if (attachments && attachments.length > 0) {
      await db.messageAttachments.createMany({
        data: attachments.map(attId => ({
          messageId: message.id,
          attachmentId: attId
        }))
      })
    }
    
    // Emit real-time event
    io.to(`user_${recipientId}`).emit('new_message', message)
    
    // Send push notification if recipient offline
    const isOnline = await checkUserOnlineStatus(recipientId)
    if (!isOnline) {
      await sendMessageNotification(recipientId, message)
    }
    
    res.json(message)
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})

// Mark messages as read
router.post('/messages/mark-read', authenticateUser, async (req, res) => {
  const userId = req.user.id
  const { messageIds } = req.body
  
  try {
    await db.messages.updateMany({
      where: {
        id: { in: messageIds },
        recipientId: userId
      },
      data: {
        status: 'read',
        readAt: new Date()
      }
    })
    
    // Notify sender of read receipts
    const messages = await db.messages.findMany({
      where: { id: { in: messageIds } }
    })
    
    messages.forEach(msg => {
      io.to(`user_${msg.senderId}`).emit('message_read', {
        messageId: msg.id,
        readAt: new Date()
      })
    })
    
    res.json({ success: true })
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})

// Upload attachment
router.post('/messages/attachments', authenticateUser, upload.single('file'), async (req, res) => {
  try {
    const file = req.file
    const userId = req.user.id
    
    const attachment = await db.attachments.create({
      data: {
        userId,
        fileName: file.originalname,
        filePath: file.path,
        fileSize: file.size,
        fileType: file.mimetype,
        uploadDate: new Date()
      }
    })
    
    res.json(attachment)
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})

// Get unread count
router.get('/messages/unread-count', authenticateUser, async (req, res) => {
  const userId = req.user.id
  
  try {
    const count = await db.messages.count({
      where: {
        recipientId: userId,
        status: { in: ['sent', 'delivered'] }
      }
    })
    
    res.json({ count })
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
})
```

## Auto-Save Draft Messages

```javascript
function useMessageDraft(conversationId) {
  const [draft, setDraft] = useState('')
  
  // Load draft on mount
  useEffect(() => {
    const savedDraft = localStorage.getItem(`draft_${conversationId}`)
    if (savedDraft) {
      setDraft(savedDraft)
    }
  }, [conversationId])
  
  // Save draft on change (debounced)
  useEffect(() => {
    const timer = setTimeout(() => {
      if (draft) {
        localStorage.setItem(`draft_${conversationId}`, draft)
      } else {
        localStorage.removeItem(`draft_${conversationId}`)
      }
    }, 500)
    
    return () => clearTimeout(timer)
  }, [draft, conversationId])
  
  const clearDraft = () => {
    setDraft('')
    localStorage.removeItem(`draft_${conversationId}`)
  }
  
  return [draft, setDraft, clearDraft]
}
```

## Performance Optimization

### Message Pagination

```javascript
function useInfiniteMessages(conversationId) {
  const [messages, setMessages] = useState([])
  const [hasMore, setHasMore] = useState(true)
  const [isLoading, setIsLoading] = useState(false)
  
  const loadMore = async () => {
    if (!hasMore || isLoading) return
    
    setIsLoading(true)
    const oldestMessage = messages[0]
    
    const response = await fetch(
      `/api/messages?before=${oldestMessage?.sentAt}&limit=20`
    )
    const data = await response.json()
    
    setMessages(prev => [...data.messages, ...prev])
    setHasMore(data.messages.length === 20)
    setIsLoading(false)
  }
  
  return { messages, loadMore, hasMore, isLoading }
}
```
